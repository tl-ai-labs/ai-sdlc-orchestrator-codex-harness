# verify-setup.mjs --enable-agent.

# Exact pin. The SDK changed what its usage totals mean between releases, so
# raise this pin only after adding the new version to AGY_USAGE_SEMANTICS in
# src/delegation/workerProcess.ts.

#

#   export DYLD_LIBRARY_PATH=/opt/homebrew/opt/expat/lib
